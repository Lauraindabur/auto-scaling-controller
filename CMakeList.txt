cmake_minimum_required(VERSION 3.13)
project(AutoScalingController CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(AWSSDK REQUIRED COMPONENTS monitoring autoscaling)

add_executable(controller
    src/main.cpp
    src/DecisionEngine.cpp
    src/MetricSource.cpp
    src/ASGActuator.cpp
    src/Logger.cpp
)

target_include_directories(controller PRIVATE include)
target_link_libraries(controller PRIVATE ${AWSSDK_LINK_LIBRARIES})
